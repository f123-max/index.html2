<script>
// --- GAME ENGINE & CONFIG ---
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');

// State
let gameState = 'START'; // START, PLAYING, GAMEOVER, CLEAR
let stage = 1;
let kills = 0;
let keys = {};
let joystickVector = { x: 0, y: 0 };
let mousePos = { x: canvas.width / 2, y: canvas.height / 2 };

// Save Data System
const SaveSystem = {
  save: function() {
    const data = { stage: stage, level: player.level, exp: player.exp, maxHp: player.maxHp };
    localStorage.setItem('last_resist_save', JSON.stringify(data));
  },
  load: function() {
    const data = localStorage.getItem('last_resist_save');
    if (data) {
      const parsed = JSON.parse(data);
      stage = parsed.stage || 1;
      player.level = parsed.level || 1;
      player.exp = parsed.exp || 0;
      player.maxHp = parsed.maxHp || 100;
      player.hp = player.maxHp;
    }
  }
};

// Player Object
const player = {
  x: 450, y: 300, radius: 15, speed: 3.5,
  hp: 100, maxHp: 100, level: 1, exp: 0, nextExp: 50,
  ultGauge: 0, maxUlt: 100,
  currentWeaponIndex: 0,
  lastShot: 0,
  angle: 0
};

// Weapons Definition
const weapons = [
  { name: 'ハンドガン', fireRate: 250, damage: 25, speed: 10, color: '#ffff00', count: 1, spread: 0 },
  { name: 'ショットガン', fireRate: 600, damage: 18, speed: 8, color: '#ffa500', count: 5, spread: 0.4 },
  { name: 'アサルトライフル', fireRate: 100, damage: 14, speed: 12, color: '#00ffff', count: 1, spread: 0.1 }
];

// Stage Definitions
const stageConfigs = [
  { name: "STAGE 1: 崩壊した都市", bg: "#1a1a24", maxKills: 20, bossSpawned: false },
  { name: "STAGE 2: 汚染研究所", bg: "#14241a", maxKills: 35, bossSpawned: false },
  { name: "STAGE 3: 最終防衛線", bg: "#241414", maxKills: 50, bossSpawned: false }
];

let bullets = [];
let enemies = [];
let particles = [];

// --- CONTROLS ---
window.addEventListener('keydown', e => { keys[e.key.toLowerCase()] = true; if(e.key.toLowerCase() === 'q') switchWeapon(); if(e.key.toLowerCase() === 'e') triggerUltimate(); });
window.addEventListener('keyup', e => { keys[e.key.toLowerCase()] = false; });
canvas.addEventListener('mousemove', e => {
  const rect = canvas.getBoundingClientRect();
  mousePos.x = (e.clientX - rect.left) * (canvas.width / rect.width);
  mousePos.y = (e.clientY - rect.top) * (canvas.height / rect.height);
});

// Touch / Joystick
const joystick = document.getElementById('joystick');
const knob = document.getElementById('knob');
let joyActive = false;
let joyStart = { x: 0, y: 0 };

joystick.addEventListener('touchstart', e => {
  joyActive = true;
  const touch = e.touches[0];
  joyStart = { x: touch.clientX, y: touch.clientY };
});
window.addEventListener('touchmove', e => {
  if (!joyActive) return;
  const touch = e.touches[0];
  const dx = touch.clientX - joyStart.x;
  const dy = touch.clientY - joyStart.y;
  const dist = Math.min(Math.hypot(dx, dy), 40);
  const angle = Math.atan2(dy, dx);
  knob.style.transform = `translate(${Math.cos(angle)*dist}px, ${Math.sin(angle)*dist}px)`;
  joystickVector = { x: Math.cos(angle) * (dist/40), y: Math.sin(angle) * (dist/40) };
});
window.addEventListener('touchend', () => {
  joyActive = false;
  knob.style.transform = `translate(0px, 0px)`;
  joystickVector = { x: 0, y: 0 };
});

document.getElementById('btn-weapon').addEventListener('click', switchWeapon);
document.getElementById('btn-ult').addEventListener('click', triggerUltimate);

function switchWeapon() {
  player.currentWeaponIndex = (player.currentWeaponIndex + 1) % weapons.length;
  document.getElementById('weapon-val').innerText = weapons[player.currentWeaponIndex].name;
}

function triggerUltimate() {
  if (player.ultGauge < player.maxUlt) return;
  player.ultGauge = 0;
  // Screen-clearing Ultimate Attack
  for (let i = 0; i < 36; i++) {
    const angle = (Math.PI * 2 / 36) * i;
    bullets.push({
      x: player.x, y: player.y,
      vx: Math.cos(angle) * 12, vy: Math.sin(angle) * 12,
      damage: 150, color: '#ffd700', radius: 6
    });
  }
  // Visual FX
  createExplosion(player.x, player.y, '#ffd700', 50);
}

// --- GAME LOGIC ---
function spawnEnemy(isBoss = false) {
  const edge = Math.floor(Math.random() * 4);
  let x, y;
  if (edge === 0) { x = Math.random() * canvas.width; y = -30; }
  else if (edge === 1) { x = canvas.width + 30; y = Math.random() * canvas.height; }
  else if (edge === 2) { x = Math.random() * canvas.width; y = canvas.height + 30; }
  else { x = -30; y = Math.random() * canvas.height; }

  enemies.push({
    x: x, y: y,
    radius: isBoss ? 35 : 12 + Math.random() * 6,
    speed: isBoss ? 1.2 : 1.5 + Math.random() * 1.5,
    hp: isBoss ? 600 * stage : 30 * stage,
    maxHp: isBoss ? 600 * stage : 30 * stage,
    color: isBoss ? '#ff0055' : (Math.random() > 0.5 ? '#e74c3c' : '#9b59b6'),
    isBoss: isBoss
  });
}

function createExplosion(x, y, color, count = 10) {
  for (let i = 0; i < count; i++) {
    particles.push({
      x: x, y: y,
      vx: (Math.random() - 0.5) * 6,
      vy: (Math.random() - 0.5) * 6,
      life: 20 + Math.random() * 10,
      color: color
    });
  }
}

function shoot() {
  const now = Date.now();
  const w = weapons[player.currentWeaponIndex];
  if (now - player.lastShot < w.fireRate) return;
  player.lastShot = now;

  let targetAngle = player.angle;
  
  // Auto-aim closest enemy on mobile if not using mouse
  if (joyActive && enemies.length > 0) {
    let closest = enemies[0];
    let minDist = Math.hypot(enemies[0].x - player.x, enemies[0].y - player.y);
    enemies.forEach(e => {
      let d = Math.hypot(e.x - player.x, e.y - player.y);
      if(d < minDist) { minDist = d; closest = e; }
    });
    targetAngle = Math.atan2(closest.y - player.y, closest.x - player.x);
  }

  for (let i = 0; i < w.count; i++) {
    const finalAngle = targetAngle + (Math.random() - 0.5) * w.spread;
    bullets.push({
      x: player.x, y: player.y,
      vx: Math.cos(finalAngle) * w.speed,
      vy: Math.sin(finalAngle) * w.speed,
      damage: w.damage, color: w.color, radius: 3
    });
  }
}

function update() {
  if (gameState !== 'PLAYING') return;

  // Movement
  let moveX = 0; let moveY = 0;
  if (keys['w'] || keys['arrowup']) moveY -= 1;
  if (keys['s'] || keys['arrowdown']) moveY += 1;
  if (keys['a'] || keys['arrowleft']) moveX -= 1;
  if (keys['d'] || keys['arrowright']) moveX += 1;

  if (joyActive) { moveX = joystickVector.x; moveY = joystickVector.y; }

  const len = Math.hypot(moveX, moveY);
  if (len > 0) {
    player.x += (moveX / (len > 1 ? len : 1)) * player.speed;
    player.y += (moveY / (len > 1 ? len : 1)) * player.speed;
  }

  // Restrict to screen
  player.x = Math.max(player.radius, Math.min(canvas.width - player.radius, player.x));
  player.y = Math.max(player.radius, Math.min(canvas.height - player.radius, player.y));

  // Player Angle
  if (!joyActive) {
    player.angle = Math.atan2(mousePos.y - player.y, mousePos.x - player.x);
  } else if (Math.hypot(moveX, moveY) > 0.1) {
    player.angle = Math.atan2(moveY, moveX);
  }

  shoot();

  // Spawning Logic
  const cfg = stageConfigs[stage - 1];
  if (kills < cfg.maxKills && enemies.length < 8 + stage * 3) {
    if (Math.random() < 0.03) spawnEnemy(false);
  } else if (kills >= cfg.maxKills && !cfg.bossSpawned) {
    cfg.bossSpawned = true;
    spawnEnemy(true);
  }

  // Update Bullets
  bullets.forEach((b, bi) => {
    b.x += b.vx; b.y += b.vy;
    if (b.x < 0 || b.x > canvas.width || b.y < 0 || b.y > canvas.height) bullets.splice(bi, 1);
  });

  // Update Enemies
  enemies.forEach((e, ei) => {
    const angle = Math.atan2(player.y - e.y, player.x - e.x);
    e.x += Math.cos(angle) * e.speed;
    e.y += Math.sin(angle) * e.speed;

    // Player collision
    if (Math.hypot(player.x - e.x, player.y - e.y) < player.radius + e.radius) {
      player.hp -= e.isBoss ? 2 : 0.5;
      if (player.hp <= 0) {
        player.hp = 0;
        gameState = 'GAMEOVER';
        document.getElementById('gameover-modal').classList.remove('hidden');
      }
    }

    // Bullet collision
    bullets.forEach((b, bi) => {
      if (Math.hypot(b.x - e.x, b.y - e.y) < b.radius + e.radius) {
        e.hp -= b.damage;
        createExplosion(b.x, b.y, b.color, 3);
        bullets.splice(bi, 1);

        if (e.hp <= 0) {
          createExplosion(e.x, e.y, e.color, e.isBoss ? 30 : 12);
          enemies.splice(ei, 1);
          kills++;
          
          // Gain EXP & ULT
          player.exp += e.isBoss ? 100 : 15;
          player.ultGauge = Math.min(player.maxUlt, player.ultGauge + (e.isBoss ? 30 : 8));

          // Level Up
          if (player.exp >= player.nextExp) {
            player.level++;
            player.exp -= player.nextExp;
            player.nextExp = Math.floor(player.nextExp * 1.4);
            player.maxHp += 15;
            player.hp = player.maxHp; // Full Heal
          }

          // Stage Clear Check
          if (e.isBoss) {
            SaveSystem.save();
            if (stage < stageConfigs.length) {
              gameState = 'CLEAR';
              document.getElementById('clear-text').innerText = `${stageConfigs[stage-1].name} 攻略完了！`;
              document.getElementById('clear-modal').classList.remove('hidden');
            } else {
              gameState = 'CLEAR';
              document.getElementById('clear-text').innerText = `全ステージクリア！世界に平和が戻った！`;
              document.getElementById('next-btn').innerText = `最初から遊ぶ`;
              document.getElementById('clear-modal').classList.remove('hidden');
            }
          }
        }
      }
    });
  });

  // Update Particles
  particles.forEach((p, pi) => {
    p.x += p.vx; p.y += p.vy; p.life--;
    if (p.life <= 0) particles.splice(pi, 1);
  });

  updateUI();
}

function updateUI() {
  document.getElementById('hp-fill').style.width = `${(player.hp / player.maxHp) * 100}%`;
  document.getElementById('hp-text').innerText = `${Math.ceil(player.hp)} / ${player.maxHp}`;
  document.getElementById('exp-fill').style.width = `${(player.exp / player.nextExp) * 100}%`;
  document.getElementById('exp-text').innerText = `${player.exp} / ${player.nextExp}`;
  document.getElementById('ult-fill').style.width = `${(player.ultGauge / player.maxUlt) * 100}%`;
  document.getElementById('ult-text').innerText = `${Math.floor((player.ultGauge / player.maxUlt) * 100)}%`;
  document.getElementById('level-val').innerText = player.level;
  document.getElementById('kill-val').innerText = kills;
}

// --- RENDER ---
function draw() {
  ctx.fillStyle = stageConfigs[stage - 1].bg;
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  // Draw Grid Effect
  ctx.strokeStyle = "rgba(255,255,255,0.03)";
  ctx.lineWidth = 1;
  for (let x = 0; x < canvas.width; x += 40) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); ctx.stroke(); }
  for (let y = 0; y < canvas.height; y += 40) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke(); }

  if (gameState === 'PLAYING' || gameState === 'CLEAR') {
    // Particles
    particles.forEach(p => {
      ctx.fillStyle = p.color;
      ctx.fillRect(p.x, p.y, 3, 3);
    });

    // Bullets
    bullets.forEach(b => {
      ctx.fillStyle = b.color;
      ctx.beginPath(); ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2); ctx.fill();
    });

    // Enemies
    enemies.forEach(e => {
      ctx.fillStyle = e.color;
      ctx.beginPath(); ctx.arc(e.x, e.y, e.radius, 0, Math.PI * 2); ctx.fill();
      // Boss Inner Ring
      if (e.isBoss) {
        ctx.strokeStyle = '#fff'; ctx.lineWidth = 3;
        ctx.beginPath(); ctx.arc(e.x, e.y, e.radius * 0.6, 0, Math.PI * 2); ctx.stroke();
      }
      // Health Bar
      ctx.fillStyle = 'rgba(0,0,0,0.5)';
      ctx.fillRect(e.x - e.radius, e.y - e.radius - 8, e.radius * 2, 4);
      ctx.fillStyle = '#ff4d4d';
      ctx.fillRect(e.x - e.radius, e.y - e.radius - 8, (e.radius * 2) * (e.hp / e.maxHp), 4);
    });

    // Player (Modern Style Vector Illustration)
    ctx.save();
    ctx.translate(player.x, player.y);
    ctx.rotate(player.angle);
    
    // Gun Body
    ctx.fillStyle = '#555';
    ctx.fillRect(5, 3, 15, 5);
    
    // Player Body
    ctx.fillStyle = '#4da6ff';
    ctx.beginPath(); ctx.arc(0, 0, player.radius, 0, Math.PI * 2); ctx.fill();
    ctx.strokeStyle = '#fff'; ctx.lineWidth = 2; ctx.stroke();

    // Player Visor / Head
    ctx.fillStyle = '#111';
    ctx.beginPath(); ctx.arc(4, 0, 8, -Math.PI/2, Math.PI/2); ctx.fill();

    ctx.restore();
  }

  requestAnimationFrame(() => { update(); draw(); });
}

// --- INITIALIZATION ---
function resetStage() {
  bullets = []; enemies = []; particles = []; kills = 0;
  player.x = canvas.width / 2; player.y = canvas.height / 2;
  stageConfigs[stage - 1].bossSpawned = false;
  document.getElementById('stage-title').innerText = stageConfigs[stage - 1].name;
}

document.getElementById('start-btn').addEventListener('click', () => {
  SaveSystem.load();
  resetStage();
  document.getElementById('start-modal').classList.add('hidden');
  gameState = 'PLAYING';
});

document.getElementById('restart-btn').addEventListener('click', () => {
  player.hp = player.maxHp;
  resetStage();
  document.getElementById('gameover-modal').classList.add('hidden');
  gameState = 'PLAYING';
});

document.getElementById('next-btn').addEventListener('click', () => {
  if (stage < stageConfigs.length) {
    stage++;
  } else {
    stage = 1; player.level = 1; player.exp = 0; player.maxHp = 100;
  }
  player.hp = player.maxHp;
  resetStage();
  document.getElementById('clear-modal').classList.add('hidden');
  gameState = 'PLAYING';
});

draw();
</script>
